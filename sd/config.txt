# Enclave OS - first boot on Raspberry Pi 1 Model B+ (BCM2835)
#
# This file is parsed by the GPU bootloader, not by our kernel, so it is kept
# pure ASCII on purpose. While debugging a completely silent UART we do not
# want a single non-standard byte anywhere on the boot path.

kernel=kernel.img

# ATAG instead of a device tree: our parser (M2) reads ATAG lists only.
# Without this line the firmware hands the kernel a flattened DTB, our parser
# truthfully reports "atags : none", and we spend a day hunting a bug that
# is not in the code.
device_tree=0

# The loader's own second-stage log on the same UART pins at 115200.
# Needed for the very first run: if these lines show up in the capture but
# ours do not, wiring and baud are fine and the bug is inside our kernel.
# Without this marker, a silent UART means "one of twenty causes".
uart_2ndstage=1

# We use no screen at all, and the GPU splash only delays handing control
# to the kernel.
disable_splash=1
